327 {

349 inline mat4 perspective(f32 fov, f32 aspect, f32 nearV, f32 farV)
350 {
351 f32 tanHalfFov = tanf(fov / 2.0f);
352 mat4 res(0.0f);
353 res.col[0][0] = 1.0f / (aspect * tanHalfFov);
354 res.col[1][1] = 1.0f / (tanHalfFov);
355 res.col[2][2] = -(farV + nearV) / (farV - nearV);
356 res.col[2][3] = -1.0f;

361 inline mat4 lookAt(const vec3& eye, const vec3& center, const vec3& up)
362 {
363 vec3 f = normalize(center - eye);
364 vec3 s = normalize(cross(f, up));
365 vec3 u = cross(s, f);
366
367 mat4 res = mat4::identity();
368 res.col[0][0] = s.x; res.col[0][1] = u.x; res.col[0][2] = -f.x;
369 res.col[1][0] = s.y; res.col[1][1] = u.y; res.col[1][2] = -f.y;
370 res.col[2][0] = s.z; res.col[2][1] = u.z; res.col[2][2] = -f.z;
371
372 res.col[3][0] = -dot(s, eye);
373 res.col[3][1] = -dot(u, eye);
374 res.col[3][2] = dot(f, eye);
375
376 return res;
